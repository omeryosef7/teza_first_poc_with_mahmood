# ---- PRIMARY ANALYSER -----------------------------------------------------------------------------
# --direction necessity flips the reference arm to NEC_BASE, inverts the capability gate to
# (point < 0 and ci95[1] < 0), inverts the rank comparator to `<=`, inverts the one-sided
# specificity alternative, skips the identity gate WITH a structured reason in
# out["gates_skipped"]["identity_check"], and writes to a NECESSITY-prefixed filename so it cannot
# overwrite the paired sufficiency report (S-120 found that it otherwise would).
#
# --control-prefixes IS MANDATORY AND MUST BE SPELLED OUT. Omit it and the default
# "KO_SHUF,KO_RAND" matches NOTHING here, the control distribution is SILENTLY EMPTY, and the
# analyser falls through to the single-comparator branch that S-050 showed can be made to say
# either thing. This is the foot-gun flagged in S-120(e).
python scripts/dcs_csi_subspace_analyze.py \
  --codeword basket --tag-prefix csi1_basket_train --expect-n 670 --allow-short 4 \
  --direction necessity \
  --arms NEC_BASE,NEC_KO,KO_NEC_FULL,KO_NEC_AXIS,KO_NEC_ORTH,KO_NEC_SHUF0,KO_NEC_SHUF1,KO_NEC_SHUF2,KO_NEC_SHUF3,KO_NEC_RAND0,KO_NEC_RAND1,KO_NEC_RAND2,KO_NEC_RAND3 \
  --base-arm NEC_BASE --ko-arm NEC_KO --full-arm KO_NEC_FULL \
  --candidate-arm KO_NEC_AXIS --comparator-arm KO_NEC_ORTH \
  --control-prefixes KO_NEC_SHUF,KO_NEC_RAND \
  --split train --require-rescue-layer 18 --require-slurm-job <HALF1>,<HALF2>

# ---- INDEPENDENT RE-DERIVATION (shares no code with the above) -------------------------------------
python scripts/dcs_csi_rederive_subspace.py \

  --direction necessity --base NEC_BASE --ko NEC_KO --full KO_NEC_FULL --candidate KO_NEC_AXIS \
  --controls KO_NEC_SHUF0,KO_NEC_SHUF1,KO_NEC_SHUF2,KO_NEC_SHUF3,KO_NEC_RAND0,KO_NEC_RAND1,KO_NEC_RAND2,KO_NEC_RAND3 \
